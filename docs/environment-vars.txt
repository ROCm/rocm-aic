# Descriptions for the generated tables in docs/ENVIRONMENT.md.
#
# scripts/gen-env-reference.py reads the defaults and owners from code and the
# text from here.  `make env-docs` regenerates the tables; `make env-docs-check`
# (and the lint workflow) fails when a variable found in code has no entry.
#
#   [section]           one table, between the `generated:<section>` markers
#   NAME: description   one row; rows keep this order.  Markdown; write '|' as '\|'.
#     continuation      an indented line continues the description
#   [ignore]            variables that are found in code but are not documented
#
# Never write a default here: the generator prints the one in code.

[image]
IMAGE_TAG: Version tag shared by the stack images, derived from `AIC_VERSION`, the pins and `ROCM_ARCH`. Set it to override the derivation; without `docker buildx` make falls back to `latest`
AIC_VERSION: AIC release. The repo-root `VERSION` file owns it (make and `aic-bake-env.sh` export it from there); the bake literal is the fallback for a bare `docker buildx bake`
AIC_IMAGE_PREFIX: Registry prefix for every image tag, with its trailing `/` (e.g. `ghcr.io/rocm/rocm-aic/`). Empty keeps the bare local names
AIC_IMAGE_EXTRA_TAG: Short alias pushed next to the version tag (e.g. `gfx1250`); only used with `AIC_IMAGE_PREFIX`
VLLM_IMAGE_REF: Full `name:tag` of the vLLM image compose starts. `make` exports the version-tagged ref bake resolves; the default is the `:latest` alias a local bake adds
LMCACHE_IMAGE_REF: Full `name:tag` of the LMCache image compose starts; resolved the same way as `VLLM_IMAGE_REF`
HSA_SNOOP_IMAGE: Full `name:tag` of the hsa-snoop image compose starts
KVBENCH_IMAGE_REF: Full `name:tag` of the KVBench image; bake tags it `aic-kvbench:<KVBENCH_IMAGE_TAG>`
KVBENCH_IMAGE_TAG: Tag of the KVBench image, the short `KVBENCH_REF`
AIC_VLLM_IMAGE: Slurm scripts: `name:tag` of the vLLM image to build, store and run. The tag is the one bake derives
AIC_LMCACHE_IMAGE: Slurm scripts: `name:tag` of the LMCache image
AIC_HSA_SNOOP_IMAGE: Slurm scripts: `name:tag` of the hsa-snoop image built by `build-exporters` and loaded by the cliff job
AIC_BASE_IMAGE: Slurm scripts: `name:tag` the `bake` command stores the base image under
AIC_IMAGE: Slurm scripts: a complete `name:tag` for the single-image paths, used verbatim; wins over `AIC_IMAGE_NAME`. The monitoring library runs its Prometheus queries from it
AIC_IMAGE_NAME: Slurm scripts: image name only; the version-derived tag is appended
AIC_EMULATE_IMAGE: `name:tag` of the CPU-only emulation image built by `make dist-build-emulate`
IMAGE_NAME: Name part of the default `AIC_EMULATE_IMAGE`
IMAGE_TAG_EMULATE: Tag part of the default `AIC_EMULATE_IMAGE`

[rocm]
ROCM_VERSION: ROCm release the images are built on
ROCM_ARCH: GPU arch of this host's stack: a `;`-, `,`- or space-separated `gfx` list. `make` fills it from `rocm_agent_enumerator` when unset and forces `gfx950` on SPUR; empty omits the arch suffix from the image tag
AIC_ROCM_ARCH: Arch list a Slurm build compiles for and a Slurm test expects (`;`-separated). Separate from `ROCM_ARCH` on purpose: the default covers every supported arch, so one image runs on any of them
AIC_FAST_ARCH: The single arch of the `-fast` make targets (`dist-build-fast`, `smoke-test-fast`, ...)
ROCM_BASE_IMAGE: Full ROCm development image the source builds run on
ROCM_RUNTIME_IMAGE: Image the shipped images start from

[pins]
PYTORCH_REF: `ROCm/pytorch` commit built from source. A SHA, because the 2.13 series has no tag
PYTORCH_SERIES: Display only: the torch series in the image tag, README and release notes, and the CPU torch the emulate builds install. Bump with `PYTORCH_REF`
TORCHVISION_REF: Ref of torchvision built against that PyTorch
VLLM_REF: vLLM ref
LLM_EMU_REF: llm-emu ref (the emulation plugin)
AITER_REF: ROCm AITER ref installed into every image; override only together with a validated `FLASH_ATTN_REF`
FLASH_ATTN_REF: FlashAttention ref, built for the CDNA (`gfx9*`) arches only
LMCACHE_REF: LMCache ref
NIXL_REF: NIXL ref
HSA_SNOOP_REF: hsa-snoop ref
CLR_REF: `ROCm/rocm-systems` ref the patched `libamdhip64` in the base image is built from; follows `ROCM_VERSION`
TRITON_REF: Triton ref source-built into the base image
UV_VERSION: `uv` release used by the image builds
NVME_EXPORTER_VERSION: `nvme_exporter` release packaged into the NVMe exporter image
RDMA_EXPORTER_VERSION: `rdma_exporter` release packaged into the RDMA exporter image
KVBENCH_REF: KVBench commit
KVBENCH_LMCACHE_VERSION: LMCache release (from `PyPI`) installed into the KVBench image. Not `LMCACHE_REF`: KVBench pins its own compatible range
AIPERF_VERSION: `aiperf` version installed into the AgentX client image

[repositories]
PYTORCH_GIT_URL: PyTorch source repository; override to build from a fork
TORCHVISION_GIT_URL: Source repository of torchvision
VLLM_GIT_URL: vLLM source repository
LLM_EMU_GIT_URL: llm-emu source repository
AITER_GIT_URL: AITER source repository
FLASH_ATTN_GIT_URL: FlashAttention source repository
LMCACHE_GIT_URL: LMCache source repository
NIXL_GIT_URL: NIXL source repository
HSA_SNOOP_GIT_URL: hsa-snoop source repository
CLR_GIT_URL: Source repository of the patched `libamdhip64`
TRITON_GIT_URL: Source repository of Triton
KVBENCH_GIT_URL: KVBench source repository

[build]
BUILD_JOBS: Cap on parallel compile jobs. Empty means `nproc`; a Slurm build defaults it to `AIC_BUILD_CPUS`
BUILD_PROGRESS: `docker buildx bake --progress` mode for `make build` (`auto`, `plain`, `tty`)
BAKE_TARGETS: Bake targets `make build` builds
BAKE_OUTPUT_FLAG: Output flag `make build` passes to bake (`--load`, `--push`)
WHEELS_DIR: Where the `wheels` bake group writes the pip wheels, one directory per wheel target; `make wheels` flattens them
VLLM_TARGET_DEVICE: vLLM build target: `rocm` compiles the GPU kernels, `empty` compiles none (the emulation image)
AIC_UCX_FAST: `1` builds a minimal UCX; `make dist-build-fast` sets it
BLAS_CACHE_BUST: Cache key of the ROCm runtime-package layer in the base image. Change it to reinstall those packages, e.g. `BLAS_CACHE_BUST=$(date +%s) make build`
AITER_PREBUILD_KERNELS: Compile the AITER kernels into the vLLM image at build time instead of on first use. `1` most, `2` no backward/tune, `3` `fmha_v3` only. Single-arch builds only
TLS_CERT: Corporate CA certificate passed to the build as a BuildKit secret, never stored in an image. A Slurm build falls back to `$HOME/certs/zscaler-ca.crt` when that file exists
AIC_BUILDX_BUILDER: Name of the `docker-container` buildx builder used by `make build` and the Slurm build; ignored with the `docker` driver
AIC_CACHE_REF: Registry ref for a shared BuildKit layer cache, stored as `<ref>-<arch>-<image>`. Takes precedence over `AIC_CACHE_GHA_SCOPE` and `AIC_CACHE_DIR`; needs `docker login`
AIC_CACHE_GHA_SCOPE: GitHub Actions cache scope prefix, for small GitHub-hosted builds only (the Actions cache is far smaller than this stack's layers)
AIC_CACHE_DIR: Directory for a file-based BuildKit layer cache, one directory per arch. With all three cache variables empty the build uses no cache
AIC_CACHE_MODE: BuildKit `--cache-to` mode: `min` or `max`

[slurm-build]
AIC_IMAGE_DIR: Shared directory the built image artifacts (tarballs or registry refs) are stored in and loaded from. The first default applies on SPUR, the alternate elsewhere
AIC_TARGETS: Comma-separated nodes the `load` and `all` commands load the images onto
AIC_DIST_BACKEND: How a build hands its images to the jobs that run them: `tarball` (`docker save` into `AIC_IMAGE_DIR`) or `registry` (a registry's storage on shared NFS plus a digest ref, so layers are stored once and a node fetches only what it lacks). `registry` needs `AIC_BUILDX_DRIVER=docker` and is the default on SPUR with that driver; `tarball` elsewhere. Nothing removes the images finished runs leave behind: run `registry garbage-collect --delete-untagged` on the storage while no job uses it
AIC_REGISTRY_IMAGE: Image of the registry the jobs start for `AIC_DIST_BACKEND=registry` and `AIC_CACHE_BACKEND=registry`. A node without it loads `<storage>/aic-images/<image>.tar`, which the first job to pull it writes
AIC_CACHE_BACKEND: Format `AIC_CACHE_DIR` is kept in: `local` (a BuildKit local cache) or `registry` (a registry's storage under `<dir>/registry`, served by a container the build job starts and removes). `registry` needs `AIC_BUILDX_DRIVER=docker` and is the default on SPUR with that driver; `local` elsewhere. Nothing removes superseded blobs from a registry: run `registry garbage-collect` while no build uses the directory
AIC_CACHE_RESET: `1` passes `reset=true` to a local `--cache-to`, so superseded blobs are dropped; needs buildx 0.35.0 or newer
AIC_CACHE_INSECURE: `1` when `AIC_CACHE_REF` presents a certificate the `docker-container` builder does not trust: it is told to skip verification
AIC_BUILDX_DRIVER: `docker-container` (a builder the script creates on the build node) or `docker` (BuildKit inside the daemon, which needs the `containerd` image store). With `docker` no builder is created or pruned and `docker/buildkitd.toml` cannot be applied, so step logs are clipped. `docker` is the default on SPUR, where the privileged builder container is denied; `docker-container` elsewhere
AIC_BUILDX_PRUNE: `1` prunes the builder's node-local cache after a `bake`; the next build compiles everything again
AIC_BUILD_PARTITION: Slurm partition for the build and load jobs. `AIC_SPUR_PARTITION` on SPUR, `defq` elsewhere
AIC_BUILD_CONSTRAINT: Slurm `-C` expression for the build node, used when `AIC_BUILD_NODE` is unset. Empty on SPUR, where nodes carry no feature labels; `CPUONLY` elsewhere
AIC_BUILD_NODE: Pin the build node with `--nodelist`; overrides `AIC_BUILD_CONSTRAINT`
AIC_BUILD_EXCLUDE_NODES: Comma-separated nodes excluded from every submit. Ignored when a node is pinned
AIC_BUILD_LOCAL: `1` builds on this host, with no Slurm job
AIC_BUILD_CPUS: `--cpus-per-task` of the build job
AIC_BUILD_MEM: `--mem` of the build job; empty leaves it to the partition default
AIC_BUILD_GPUS: `--gpus` of the build job. SPUR refuses a submit with no GPU request, so it defaults to `1` there; unset elsewhere
AIC_BUILD_TIME: Build job time limit
AIC_LOAD_TIME: Per-node load job time limit
AIC_TMP_DIR: Build-job `TMPDIR` and OCI scratch. Empty picks node-local `/tmp/aic-scratch.<job>`, then `$HOME/.tmp-rocm-aic-cicd`; never shared NFS
AIC_COMPRESS: Tarball compression: `zstd`, `gzip` or `none`. Empty picks `zstd` when available, else `gzip`
AIC_BUILD_TARGET: Dockerfile stage the single-image `build` command stops at (`emulate` for the emulation image)
AIC_BUILD_DOCKERFILE: Dockerfile (relative to `docker/`) the single-image `build` command builds
AIC_BUILD_CONTEXT_BASE: Pre-built base image handed to that build as its `base` stage
AIC_BUILD_BASE_FIRST: `1` builds the base image first, in the same job, instead
AIC_BUILD_CONTEXT_BASE_WHEELS: Directory of exported base wheels handed to that build as its `base-wheels` context
AIC_BUILD_CONTEXT_LMCACHE_WHEELS: Directory of exported LMCache wheels handed to that build as its `lmcache-wheels` context
AIC_VLLM_TARGET_DEVICE: Slurm name for bake's `VLLM_TARGET_DEVICE` in the single-image build
AIC_EMULATE_VLLM_DEVICE: `VLLM_TARGET_DEVICE` for `build-emulate`; empty takes the emulate target's `empty`
AIC_AITER_PREBUILD_KERNELS: Slurm name for bake's `AITER_PREBUILD_KERNELS`; needs a single-arch `AIC_ROCM_ARCH`
AIC_WHEEL_DIR: Where the `wheels` command exports the pip wheels
AIC_VLLM_PUSH_REF: Registry-qualified ref `make dist-push` pushes the vLLM image to. Both push refs are required
AIC_LMCACHE_PUSH_REF: Registry-qualified ref `make dist-push` pushes the LMCache image to
AIC_FORCE_LOAD: `1` reloads an image from its tarball even when the node's copy is current. By default a node reloads only when the tarball is newer than what it last loaded
AIC_REQUIRE_FRESH_TARBALL: `1` fails a build whose tarball was not rewritten, instead of warning
AIC_JOB_LOG_TAIL: Lines of the job log printed when a submitted job fails
AIC_SLURM_ACCOUNT: Slurm `--account` for every submit; not needed on SPUR

[site]
AIC_CLUSTER: Selects the per-cluster environment file `.env.<cluster>` in the repo root. Auto-detected: the first dash-separated token of the host name with a matching file
AIC_ENV_FILE: Path of that file, overriding the `AIC_CLUSTER` lookup. Plain unquoted `KEY=value` lines
AIC_SPUR_CLUSTER: `1` on a SPUR controller: selects the SPUR site defaults shown in these tables and the SPUR submit path. SPUR's `sbatch` has no `--parsable`, `--wait` or `--overcommit`, so jobs are submitted from a file and watched with `squeue`
AIC_SPUR_PARTITION: Partition for every SPUR submit. Use the variable, not the literal name
AIC_SHARED_NFS: Root of the shared NFS on SPUR, under which the image, cache and Hugging Face directories default
AIC_SPUR_RPC_RETRIES: Consecutive `squeue` failures tolerated while watching a job (10 s apart) before giving up
AIC_CI_LIB_DIR: Where `make install-ci-scripts` installs the self-hosted runner scripts
AIC_CI_STORAGE_ROOT: Root of the CI checkouts on the SPUR login node
AIC_CI_ACTIVE_JOB_FILE: CI runners: file the id of the job being watched is written to, so a cancelled workflow can cancel it

[gpu-storage]
GPU: Fallback GPU index when no scheduler allocation is resolved; also names the vLLM container (`aic-vllm-gpu<GPU>`)
AIC_ROCR_VISIBLE: `ROCR_VISIBLE_DEVICES` for every container with `/dev/kfd`: absolute host GPU ids (`3` or `2,5`). The Slurm jobs resolve it: on standard Slurm from `ROCR_VISIBLE_DEVICES`, then `HIP_VISIBLE_DEVICES`, then `CUDA_VISIBLE_DEVICES`; on SPUR from the controller allocation, and an allocation that cannot be resolved fails the job
AIC_HIP_VISIBLE: `HIP_VISIBLE_DEVICES` and `CUDA_VISIBLE_DEVICES`: indices relative to `AIC_ROCR_VISIBLE` (`0` or `0,1`). Set the two together
NVME_DATA: Host path of the local NVMe pool (L2a), mounted at `/data/nvme`. The cliff job picks its own under `AIC_LOCAL_ROOT`
NFS_DATA: Host path of the NFS-over-RDMA pool (L2b), mounted at `/data/nfs`
GDS_SLAB_DATA: Host path of the hipFile slab file; required with `GDS_MODE=1`
GDS_MODE: `1` uses the hipFile NVMe slab as L1 instead of DRAM (`make up-gds-l1`)
AIC_SHM_SIZE: Size of the private `/dev/shm` shared by LMCache and vLLM; raise it when the L1 request is larger

[hf-logging]
HF_TOKEN: Hugging Face access token. Required to download gated models; not needed for an offline cache
HF_TOKEN_FILE: File holding the token, read by make when `HF_TOKEN` is unset
HF_HOME: Host Hugging Face cache directory mounted into the containers. `make` uses `/scratch/models` when that exists; each Slurm job has its own default
HF_HUB_OFFLINE: `1` blocks Hugging Face Hub network access
TRANSFORMERS_OFFLINE: `1` puts Transformers in offline mode
LOG: Host log directory
TZ: Time zone of every container

[lmcache]
LMCACHE_PORT: ZMQ listen port of the LMCache MP server
LMCACHE_L1_SIZE_GB: L1 cap in GiB: DRAM in the standard mode, slab size in GDS mode
AIC_L2_BACKEND: L2 backend: `nixl` (AIS_MT NVMe plus POSIX NFS), `nixl_posix`, `nixl_posix_dynamic`, `local_disk` (LMCache's own disk backend through a mounted config) or `none`. `nixl_posix` is the SPUR and emulation default
AIC_NIXL_BACKEND: NIXL plugin of the `nixl` backend: `AIS_MT` (hipFile, P2PDMA) or `POSIX`
AIC_NIXL_USE_DIRECT_IO: `O_DIRECT` for the AIS_MT NVMe pool. The POSIX plugin always uses buffered I/O
LMCACHE_NVME_POOL: NIXL pool slots of the NVMe adapter
LMCACHE_NVME_SLOT_SIZE: File size per NVMe pool slot, bytes (256 MiB)
LMCACHE_NFS_POOL: NIXL pool slots of the NFS adapter
LMCACHE_NIXL_POSIX_SLOT_SIZE: Size of each POSIX pool file, bytes (16 MiB). L2 capacity is pool x this size; a multiple of 4 KiB
LMCACHE_NIXL_POSIX_POOL: Pool slots of the POSIX backend
LMCACHE_NIXL_DYNAMIC_MAX_GB: NVMe capacity cap in GB for `nixl_posix_dynamic`
LMCACHE_NIXL_DYNAMIC_DIRECT_IO: `O_DIRECT` for `nixl_posix_dynamic`
LMCACHE_MAX_GPU_WORKERS: `--max-gpu-workers`, the parallel store and retrieve workers. `1` puts every client on one thread and backs stores up under concurrent load; use it only for a single-client run, to save GPU memory
LMCACHE_WORKER_REAP_TIMEOUT: `--worker-reap-timeout-seconds`
LMCACHE_L2_PREFETCH_MAX_IN_FLIGHT: `--l2-prefetch-max-in-flight`, concurrent L2 load tasks. The cliff job raises it
LMCACHE_CHUNK_SIZE: Tokens per KV chunk; the MP server and the coordinator must agree
LMCACHE_CONFIG_FILE_HOST: Host path of an LMCache config file to mount (`local_disk` and advanced tuning)
LMCACHE_CONFIG_FILE: In-container path of that config; set only when one is mounted
NIXL_TELEMETRY_ENABLE: NIXL's own Prometheus exporter on `:19090`. Set it empty to disable
NIXL_TELEMETRY_EXPORTER: NIXL telemetry exporter type
AIC_HSA_IPC_LEGACY: `1` sets `HSA_ENABLE_IPC_MODE_LEGACY=1` in both containers: the KFD-handle IPC mode, which needs the DKMS `amdgpu` module
AIC_LMCACHE_VMM_IPC: `1` selects the HIP VMM wrapper for cross-container memory handles

[coordinator]
LMCACHE_COORDINATOR_URL: Base URL the MP server registers and heartbeats against. Set it empty to run without a coordinator
LMCACHE_COORDINATOR_INSTANCE_TIMEOUT: Seconds without a heartbeat before an instance is evicted
LMCACHE_COORDINATOR_HEALTH_INTERVAL: Seconds between heartbeat sweeps
LMCACHE_COORDINATOR_EVICTION_INTERVAL: Seconds between L2 eviction checks
LMCACHE_COORDINATOR_KEEPALIVE: HTTP keep-alive timeout, seconds

[vllm]
VLLM_MODEL: Model to serve. The cliff job picks one by arch when it is empty (`AIC_MODEL_BIG`, `AIC_MODEL_SMALL`)
TENSOR_PARALLEL_SIZE: `--tensor-parallel-size`
VLM_GPU_MEMORY_UTILIZATION: `--gpu-memory-utilization`
VLM_MAX_MODEL_LEN: `--max-model-len`
VLM_MAX_NUM_BATCHED_TOKENS: `--max-num-batched-tokens`
VLM_BLOCK_SIZE: `--block-size`; match it to the model and connector cache geometry
VLM_LOAD_FORMAT: `--load-format`. `fastsafetensors` is CDNA-only; use `auto` on RDNA
VLM_ATTENTION_BACKEND: `--attention-backend`. `TRITON_ATTN` is the one that supports KV connectors
VLM_KV_CACHE_DTYPE: `--kv-cache-dtype`; `auto` for arches without native `fp8`
VLM_KV_CACHE_METRICS_SAMPLE: Fraction of KV blocks sampled for the residency histograms
KV_TRANSFER_ARG: vLLM `--kv-transfer-config` argument. `make` sets the `LMCacheMPConnector` JSON; empty is a plain vLLM (the cliff `vram` baseline). Single-quote the JSON
VLLM_EXTRA_ARGS: Extra `vllm serve` arguments appended verbatim; single-quote embedded JSON
VLLM_LOGGING_LEVEL: vLLM log level
VLLM_SERVER_DEV_MODE: `1` enables vLLM's development endpoints, which the tests use to reset the GPU prefix cache
AIC_ROCM_USE_AITER: `VLLM_ROCM_USE_AITER`: the AITER unified attention a KV connector needs. `0` for arches without AITER support
PYTORCH_ALLOC_CONF: Passed through to vLLM. A no-op on ROCm; see `PYTORCH_HIP_ALLOC_CONF`
PYTORCH_HIP_ALLOC_CONF: PyTorch HIP allocator configuration for vLLM
VLLM_IPC_MODE: IPC namespace of the `vllm` service. It joins LMCache's; to run vLLM alone start the `vllm-only` service rather than overriding this
VLLM_PID_MODE: PID namespace of the `vllm` service. Empty means a private namespace
VLLM_NETWORK_MODE: Network namespace of the `vllm` service. ROCR passes IPC handles over an abstract socket, so vLLM and LMCache must share one
VLM_READY_RETRIES: `make vllm-reset-test`: readiness polls, 5 s apart
VLM_FAIL_LOG_LINES: `make vllm-reset-test`: container log lines printed when the vLLM container exits before becoming ready

[network]
AIC_PUBLISH_ADDR: Host address the published Grafana port binds; `0.0.0.0` exposes it beyond loopback
AIC_GRAFANA_PORT: Pin the Grafana host port, the only published one. Empty lets Docker pick a free port; `make endpoints` prints the URL
AIC_CLIENT_IMAGE: Image of the `aic-client` container make targets run their clients in
AIC_FETCH_IMAGE: Image `aic_fetch` runs `curl` from; scripts set it to an AIC image already on the node
AIC_FETCH_TIMEOUT: `aic_fetch` request timeout, seconds
AIC_NETWORK: Docker network `aic_fetch` and the endpoint helpers attach to

[monitoring]

AIC_METRICS_DIR: Prometheus TSDB directory. The cliff job writes it under its own log directory
AIC_PROM_IMAGE: Prometheus image
AIC_PROM_RETENTION: Prometheus TSDB retention
PROM_UID: User id Prometheus runs as, so it can write the TSDB directory; make passes the caller's
PROM_GID: Group id Prometheus runs as
AIC_GRAFANA_IMAGE: Grafana image
AIC_NODE_EXPORTER_IMAGE: Node exporter image
AIC_AMDGPU_EXPORTER_IMAGE: AMD GPU metrics exporter image
AIC_NVME_EXPORTER_IMAGE: NVMe exporter image. A local bake tags it `:local`; the Slurm build stores it as `:latest`
AIC_RDMA_EXPORTER_IMAGE: RDMA exporter image; tagged like the NVMe one
AIC_HSA_SNOOP_PID_MODE: PID namespace of the hsa-snoop container. Empty means private. The monitoring library defaults it to `host`; SPUR refuses that, so make sets `container:aic-lmcache` there
NODEX_PID_MODE: PID namespace of the node exporter; `host` gives per-process CPU statistics where the site permits it
AIC_CPU_SMOKE: `1` starts the CPU-only exporter subset (the monitoring smoke test on a node with no GPU)
AIC_MON_SKIP_SERVICES: Space-separated compose services the monitoring library leaves out
COMPOSE_PLUGIN_VERSION: `docker compose` plugin version installed by `make ensure-compose` and the Slurm jobs when the plugin is missing
PROM_DUMP_OUT: Output path of `make prometheus-dump`
PROM_DUMP_WAIT: `make prometheus-dump`: seconds between the stack coming up and the scrape

[bench]
BENCH_ARM: Arm `make cliff` runs: `kvd_v2` or `vram_only`

BENCH_PREFIX_MODE: Prefix sharing in the Slurm cliff job: `per_client` or `shared`. `make cliff-kvd` presets `shared`
BENCH_LOGDIR: Where non-Slurm runs write results and plots

[cliff]
AIC_CLIFF_ARMS: Comma list of arms to run: `vram`, `nvme`, `gds`. `make cliff-spur-l2` presets `nvme`
AIC_CLIFF_NODE: Pin the cliff node with `--nodelist`

AIC_CLIFF_CPUS: `--cpus-per-task` of the cliff job
AIC_CLIFF_MEM: `--mem` of the cliff job
AIC_CLIFF_JOB_NAME: Slurm job name; the make cliff targets each set their own
